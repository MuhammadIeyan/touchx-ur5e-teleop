cmake_minimum_required(VERSION 3.16)
project(sim_contact_feedback)
find_package(ament_cmake REQUIRED)
find_package(mujoco_ros2_control_plugins REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(pluginlib REQUIRED)
add_library(sim_contact_feedback SHARED contact_feedback.cpp)
target_compile_features(sim_contact_feedback PUBLIC cxx_std_17)
target_compile_options(sim_contact_feedback PRIVATE -Wall -Wextra)
target_link_libraries(sim_contact_feedback PUBLIC mujoco_ros2_control_plugins::mujoco_ros2_control_plugins ${geometry_msgs_TARGETS} pluginlib::pluginlib)
install(TARGETS sim_contact_feedback LIBRARY DESTINATION lib)
pluginlib_export_plugin_description_file(mujoco_ros2_control_plugins plugins.xml)
add_executable(test_contact test_contact.cpp)
target_link_libraries(test_contact PUBLIC mujoco_ros2_control_plugins::mujoco_ros2_control_plugins)
enable_testing()
add_test(NAME contact_direction COMMAND test_contact)

ament_package()
