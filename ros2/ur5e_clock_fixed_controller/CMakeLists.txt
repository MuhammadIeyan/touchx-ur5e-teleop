cmake_minimum_required(VERSION 3.16)
project(ur5e_clock_fixed_controller)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ur_controllers REQUIRED)
find_package(pluginlib REQUIRED)
add_library(${PROJECT_NAME} SHARED src/scaled_joint_trajectory_controller.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} ur_controllers pluginlib)
pluginlib_export_plugin_description_file(controller_interface controller_plugins.xml)
add_executable(clock_fixed_offline_check src/offline_check.cpp)
target_link_libraries(clock_fixed_offline_check ${PROJECT_NAME})
ament_target_dependencies(clock_fixed_offline_check ur_controllers)
add_executable(vendor_offline_check src/offline_check.cpp)
target_compile_definitions(vendor_offline_check PRIVATE CHECK_VENDOR)
ament_target_dependencies(vendor_offline_check ur_controllers)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_package()
